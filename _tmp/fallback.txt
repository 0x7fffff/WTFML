Quote fallbackFor(Script script, quint32 day) {
    QVector<Quote> same;
    for (const Quote &quote : fallbackQuotes())
        if (quote.script == script) same.append(quote);
    if (same.isEmpty()) return Quote{};
    return same.at(int(day % quint32(same.size())));
}
